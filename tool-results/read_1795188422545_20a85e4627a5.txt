   158→   158→function useRevealOnScroll() {
   159→   159→  useEffect(() => {
   160→   160→    const els = Array.from(document.querySelectorAll(".rv"));
   161→   161→    if (!("IntersectionObserver" in window)) {
   162→   162→      els.forEach((el) => el.classList.add("is-in"));
   163→   163→      return;
   164→   164→    }
   165→   165→    const io = new IntersectionObserver(
   166→   166→      (entries) => {
   167→   167→        for (const e of entries) {
   168→   168→          if (e.isIntersecting) {
   169→   169→            e.target.classList.add("is-in");
   170→   170→            io.unobserve(e.target);
   171→   171→          }
   172→   172→        }
   173→   173→      },
   174→   174→      { threshold: 0.12, rootMargin: "0px 0px -6% 0px" },
   175→   175→    );
   176→   176→    els.forEach((el) => io.observe(el));
   177→   177→    return () => io.disconnect();
   178→   178→  }, []);
   179→   179→}

   181→   181→function CountUp({ to, suffix, duration = 1600 }: { to: number; suffix?: string; duration?: number }) {
   182→   182→  const ref = useRef<HTMLSpanElement>(null);
   183→   183→  const [val, setVal] = useState(0);
   184→   184→  const started = useRef(false);
   185→   185→  useEffect(() => {
   186→   186→    const el = ref.current;
   187→   187→    if (!el) return;
   188→   188→    const io = new IntersectionObserver(
   189→   189→      (entries) => {
   190→   190→        if (!entries[0].isIntersecting || started.current) return;
   191→   191→        started.current = true;
   192→   192→        const t0 = performance.now();
   193→   193→        const tick = (t: number) => {
   194→   194→          const p = Math.min(1, (t - t0) / duration);
   195→   195→          const eased = 1 - Math.pow(1 - p, 3);
   196→   196→          setVal(Math.round(to * eased));
   197→   197→          if (p < 1) requestAnimationFrame(tick);
   198→   198→        };
   199→   199→        requestAnimationFrame(tick);
   200→   200→      },
   201→   201→      { threshold: 0.4 },
   202→   202→    );
   203→   203→    io.observe(el);
   204→   204→    return () => io.disconnect();
   205→   205→  }, [to, duration]);
   206→   206→  return (
   207→   207→    <span ref={ref}>
   208→   208→      {val}
   209→   209→      {suffix}
   210→   210→    </span>
   211→   211→  );
   212→   212→}
