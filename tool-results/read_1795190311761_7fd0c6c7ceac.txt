   154→function useRevealOnScroll() {
   155→  useEffect(() => {
   156→    const els = Array.from(document.querySelectorAll(".rv"));
   157→    if (!("IntersectionObserver" in window)) {
   158→      els.forEach((el) => el.classList.add("is-in"));
   159→      return;
   160→    }
   161→    const io = new IntersectionObserver(
   162→      (entries) => {
   163→        for (const e of entries) {
   164→          if (e.isIntersecting) {
   165→            e.target.classList.add("is-in");
   166→            io.unobserve(e.target);
   167→          }
   168→        }
   169→      },
   170→      { threshold: 0.12, rootMargin: "0px 0px -6% 0px" },
   171→    );
   172→    els.forEach((el) => io.observe(el));
   173→    return () => io.disconnect();
   174→  }, []);
   175→}

   177→function CountUp({ to, suffix, duration = 1600 }: { to: number; suffix?: string; duration?: number }) {
   178→  const ref = useRef<HTMLSpanElement>(null);
   179→  const [val, setVal] = useState(0);
   180→  const started = useRef(false);
   181→  useEffect(() => {
   182→    const el = ref.current;
   183→    if (!el) return;
   184→    const io = new IntersectionObserver(
   185→      (entries) => {
   186→        if (!entries[0].isIntersecting || started.current) return;
   187→        started.current = true;
   188→        const t0 = performance.now();
   189→        const tick = (t: number) => {
   190→          const p = Math.min(1, (t - t0) / duration);
   191→          const eased = 1 - Math.pow(1 - p, 3);
   192→          setVal(Math.round(to * eased));
   193→          if (p < 1) requestAnimationFrame(tick);
   194→        };
   195→        requestAnimationFrame(tick);
   196→      },
   197→      { threshold: 0.4 },
   198→    );
   199→    io.observe(el);
   200→    return () => io.disconnect();
   201→  }, [to, duration]);
   202→  return (
   203→    <span ref={ref}>
   204→      {val}
   205→      {suffix}
   206→    </span>
   207→  );
   208→}
