   158→function useRevealOnScroll() {
   159→  useEffect(() => {
   160→    const els = Array.from(document.querySelectorAll(".rv"));
   161→    if (!("IntersectionObserver" in window)) {
   162→      els.forEach((el) => el.classList.add("is-in"));
   163→      return;
   164→    }
   165→    const io = new IntersectionObserver(
   166→      (entries) => {
   167→        for (const e of entries) {
   168→          if (e.isIntersecting) {
   169→            e.target.classList.add("is-in");
   170→            io.unobserve(e.target);
   171→          }
   172→        }
   173→      },
   174→      { threshold: 0.12, rootMargin: "0px 0px -6% 0px" },
   175→    );
   176→    els.forEach((el) => io.observe(el));
   177→    return () => io.disconnect();
   178→  }, []);
   179→}

   181→function CountUp({ to, suffix, duration = 1600 }: { to: number; suffix?: string; duration?: number }) {
   182→  const ref = useRef<HTMLSpanElement>(null);
   183→  const [val, setVal] = useState(0);
   184→  const started = useRef(false);
   185→  useEffect(() => {
   186→    const el = ref.current;
   187→    if (!el) return;
   188→    const io = new IntersectionObserver(
   189→      (entries) => {
   190→        if (!entries[0].isIntersecting || started.current) return;
   191→        started.current = true;
   192→        const t0 = performance.now();
   193→        const tick = (t: number) => {
   194→          const p = Math.min(1, (t - t0) / duration);
   195→          const eased = 1 - Math.pow(1 - p, 3);
   196→          setVal(Math.round(to * eased));
   197→          if (p < 1) requestAnimationFrame(tick);
   198→        };
   199→        requestAnimationFrame(tick);
   200→      },
   201→      { threshold: 0.4 },
   202→    );
   203→    io.observe(el);
   204→    return () => io.disconnect();
   205→  }, [to, duration]);
   206→  return (
   207→    <span ref={ref}>
   208→      {val}
   209→      {suffix}
   210→    </span>
   211→  );
   212→}
